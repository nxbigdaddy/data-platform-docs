```python
# core/base_model.py
from datetime import datetime
from typing import Optional
from sqlalchemy import Column, BigInteger, DateTime, SmallInteger
from sqlalchemy.orm import declarative_base

Base = declarative_base()


class BaseModel(Base):
    """基础模型类"""
    __abstract__ = True
    
    id = Column(BigInteger, primary_key=True, autoincrement=True)
    created_at = Column(DateTime, default=datetime.now, nullable=False)
    updated_at = Column(DateTime, default=datetime.now, onupdate=datetime.now, nullable=False)
    created_by = Column(BigInteger)
    updated_by = Column(BigInteger)
    is_deleted = Column(SmallInteger, default=0, nullable=False)
    
    def to_dict(self, exclude: Optional[list] = None) -> dict:
        """转换为字典"""
        exclude = exclude or []
        result = {}
        for column in self.__table__.columns:
            if column.name not in exclude:
                value = getattr(self, column.name)
                if isinstance(value, datetime):
                    value = value.isoformat()
                result[column.name] = value
        return result


# core/base_service.py
from typing import TypeVar, Generic, List, Optional
from sqlalchemy.orm import Session

T = TypeVar('T', bound=BaseModel)


class BaseService(Generic[T]):
    """基础服务类"""
    
    def __init__(self, db: Session, model_class: type[T]):
        self.db = db
        self.model_class = model_class
    
    def get_by_id(self, id: int) -> Optional[T]:
        """根据ID获取"""
        return self.db.query(self.model_class).filter(
            self.model_class.id == id,
            self.model_class.is_deleted == 0
        ).first()
    
    def list(self, skip: int = 0, limit: int = 100, **filters) -> List[T]:
        """列表查询"""
        query = self.db.query(self.model_class).filter(self.model_class.is_deleted == 0)
        
        for key, value in filters.items():
            if value is not None and hasattr(self.model_class, key):
                query = query.filter(getattr(self.model_class, key) == value)
        
        return query.offset(skip).limit(limit).all()
    
    def create(self, obj_in: dict, user_id: Optional[int] = None) -> T:
        """创建"""
        db_obj = self.model_class(**obj_in)
        if user_id:
            db_obj.created_by = user_id
        self.db.add(db_obj)
        self.db.commit()
        self.db.refresh(db_obj)
        return db_obj
    
    def update(self, id: int, obj_in: dict, user_id: Optional[int] = None) -> Optional[T]:
        """更新"""
        db_obj = self.get_by_id(id)
        if not db_obj:
            return None
        
        for key, value in obj_in.items():
            if hasattr(db_obj, key):
                setattr(db_obj, key, value)
        
        if user_id:
            db_obj.updated_by = user_id
        
        self.db.commit()
        self.db.refresh(db_obj)
        return db_obj
    
    def delete(self, id: int, user_id: Optional[int] = None, soft: bool = True) -> bool:
        """删除"""
        db_obj = self.get_by_id(id)
        if not db_obj:
            return False
        
        if soft:
            db_obj.is_deleted = 1
            if user_id:
                db_obj.updated_by = user_id
            self.db.commit()
        else:
            self.db.delete(db_obj)
            self.db.commit()
        return True
```

```python
# infrastructure/connectors/base_connector.py
from abc import ABC, abstractmethod
from typing import List, Dict, Any, Optional


class BaseConnector(ABC):
    """数据源连接器基类"""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.connection = None
    
    @abstractmethod
    def connect(self):
        """建立连接"""
        pass
    
    @abstractmethod
    def disconnect(self):
        """断开连接"""
        pass
    
    @abstractmethod
    def test_connection(self) -> bool:
        """测试连接"""
        pass
    
    @abstractmethod
    def get_schemas(self) -> List[str]:
        """获取Schema列表"""
        pass
    
    @abstractmethod
    def get_tables(self, schema: Optional[str] = None) -> List[Dict]:
        """获取表列表"""
        pass
    
    @abstractmethod
    def get_columns(self, table: str, schema: Optional[str] = None) -> List[Dict]:
        """获取字段列表"""
        pass
    
    @abstractmethod
    def execute_query(self, query: str, params: Optional[Dict] = None) -> List[Dict]:
        """执行查询"""
        pass
    
    @abstractmethod
    def get_table_data(self, table: str, schema: Optional[str] = None, 
                       limit: int = 100) -> List[Dict]:
        """获取表数据样本"""
        pass

        finally:

            with self.connection.cursor() as cursor:
                database = schema or self.config.get('database')
                cursor.execute("""

                        TABLE_COMMENT as table_comment,
                        TABLE_ROWS as row_count,
                        DATA_LENGTH as data_size
                    FROM INFORMATION_SCHEMA.TABLES
                    WHERE TABLE_SCHEMA = %s
                """, (database,))
                return cursor.fetchall()
        finally:
            self.disconnect()
    
    def get_columns(self, table: str, schema: Optional[str] = None) -> List[Dict]:
        self.connect()
        try:
            with self.connection.cursor() as cursor:
                database = schema or self.config.get('database')
                cursor.execute("""
                    SELECT 
                        COLUMN_NAME as column_name,
                        COLUMN_COMMENT as column_comment,
                        DATA_TYPE as data_type,
                        CHARACTER_MAXIMUM_LENGTH as column_length,
                        NUMERIC_PRECISION as column_precision,
                        NUMERIC_SCALE as column_scale,
                        IS_NULLABLE as is_nullable,
                        COLUMN_DEFAULT as default_value,
                        ORDINAL_POSITION as ordinal_position
                    FROM INFORMATION_SCHEMA.COLUMNS
                    WHERE TABLE_SCHEMA = %s AND TABLE_NAME = %s
                    ORDER BY ORDINAL_POSITION
                """, (database, table))
                return cursor.fetchall()
        finally:
            self.disconnect()
    
    def execute_query(self, query: str, params: Optional[Dict] = None) -> List[Dict]:
        self.connect()
        try:
            with self.connection.cursor() as cursor:
                cursor.execute(query, params)
                return cursor.fetchall()
        finally:
            self.disconnect()
    
    def get_table_data(self, table: str, schema: Optional[str] = None, 
                       limit: int = 100) -> List[Dict]:
        database = schema or self.config.get('database')
        query = f"SELECT * FROM {database}.{table} LIMIT {limit}"
        return self.execute_query(query)
```

```python
    """血缘边"""
    source: str
    target: str
    relation: str  # direct/derived/aggregate


class SQLLineageAnalyzer:
    """SQL血缘分析器"""
    
    # SQL关键字
    DDL_KEYWORDS = ['CREATE', 'ALTER', 'DROP', 'TRUNCATE']
    DML_KEYWORDS = ['INSERT', 'UPDATE', 'DELETE', 'MERGE']
    QUERY_KEYWORDS = ['SELECT', 'FROM', 'JOIN', 'WHERE', 'GROUP BY', 'ORDER BY']
    
    def __init__(self):
        self.tables = {}
        self.columns = {}
        self.edges = []
    
    def analyze(self, sql: str) -> Dict:
        """分析SQL血缘关系"""
        sql = sql.upper().strip()
        
        # 判断SQL类型
        if any(sql.startswith(kw) for kw in self.DDL_KEYWORDS):
            return self._analyze_ddl(sql)
        elif any(sql.startswith(kw) for kw in self.DML_KEYWORDS):
            return self._analyze_dml(sql)
        elif sql.startswith('SELECT'):
            return self._analyze_query(sql)
        else:
            return {"type": "unknown", "nodes": [], "edges": []}
    
    def _analyze_query(self, sql: str) -> Dict:
        """分析查询SQL"""
        # 提取源表
        source_tables = self._extract_tables(sql)
        
        # 提取字段映射
        column_mappings = self._extract_column_mappings(sql)
        
        # 提取目标表（如果是CTAS）
        target_table = self._extract_target_table(sql)
        
        nodes = []
        edges = []
        
        # 添加源表节点
        for table in source_tables:
            node_id = f"table:{table}"
            nodes.append({
                "id": node_id,
                "name": table,
                "type": "table"
            })
        
        # 添加目标表节点
        if target_table:
            target_id = f"table:{target_table}"
            nodes.append({
                "id": target_id,
                "name": target_table,
                "type": "table"
            })
            
            # 添加表级血缘边
            for table in source_tables:
                edges.append({
                    "source": f"table:{table}",
                    "target": target_id,
                    "relation": "derived"
                })
        
        return {
            "type": "query",
            "nodes": nodes,
            "edges": edges,
            "column_mappings": column_mappings
        }
    
    def _extract_tables(self, sql: str) -> List[str]:
        """从SQL中提取表名"""
        tables = []
        
        # FROM子句
        from_pattern = r'FROM\s+(\w+)(?:\s+AS\s+\w+)?'
        tables.extend(re.findall(from_pattern, sql, re.IGNORECASE))
        
        # JOIN子句
        join_pattern = r'JOIN\s+(\w+)(?:\s+AS\s+\w+)?'
        tables.extend(re.findall(join_pattern, sql, re.IGNORECASE))
        
        return list(set(tables))
    
    def _extract_column_mappings(self, sql: str) -> List[Dict]:
        """提取字段映射关系"""
        mappings = []
        
        # 提取SELECT子句
        select_match = re.search(r'SELECT\s+(.+?)\s+FROM', sql, re.IGNORECASE | re.DOTALL)
        if select_match:
            select_clause = select_match.group(1)
            
            # 分割字段
            columns = self._split_columns(select_clause)
            
            for col in columns:
                # 解析字段别名
                col = col.strip()
                if ' AS ' in col.upper():
                    parts = re.split(r'\s+AS\s+', col, flags=re.IGNORECASE)
                    source = parts[0].strip()
                    target = parts[1].strip()
                    mappings.append({
                        "source": source,
                        "target": target
                    })
                else:
                    mappings.append({
                        "source": col,
                        "target": col
                    })
        
        return mappings
    
    def _split_columns(self, select_clause: str) -> List[str]:
        """分割SELECT字段"""
        columns = []
        current = ""
        paren_count = 0
        
        for char in select_clause:
            if char == '(':
                paren_count += 1
            elif char == ')':
                paren_count -= 1
            elif char == ',' and paren_count == 0:
                columns.append(current.strip())
                current = ""
                continue
            current += char
        
        if current.strip():
            columns.append(current.strip())
        
        return columns
    
    def _extract_target_table(self, sql: str) -> str:
        """提取目标表名（CTAS）"""
        # CREATE TABLE AS SELECT
        ctas_pattern = r'CREATE\s+(?:OR\s+REPLACE\s+)?TABLE\s+(\w+)\s+AS'
        match = re.search(ctas_pattern, sql, re.IGNORECASE)
        if match:
            return match.group(1)
        
        # INSERT INTO
        insert_pattern = r'INSERT\s+(?:INTO\s+)?(\w+)'
        match = re.search(insert_pattern, sql, re.IGNORECASE)
        if match:
            return match.group(1)
        
        return None
```

### 4.2 数据脱敏算法

```python
# apps/security/masking/masking_engine.py
import hashlib
import re
import random
from typing import Dict, Any, Optional
from enum import Enum


class MaskingType(Enum):
    """脱敏类型"""
    MASK = "mask"           # 掩码
    HASH = "hash"           # 哈希
    ENCRYPT = "encrypt"     # 加密
    REPLACE = "replace"     # 替换
    TRUNCATE = "truncate"   # 截断


class MaskingEngine:
    """脱敏引擎"""
    
    def __init__(self):
        self.rules = {}
    
    def mask(self, value: Any, rule_config: Dict) -> Any:
        """执行脱敏"""
        if value is None:
            return None
        
        masking_type = MaskingType(rule_config.get('type', 'mask'))
        
        if masking_type == MaskingType.MASK:
            return self._mask(value, rule_config)
        elif masking_type == MaskingType.HASH:
            return self._hash(value, rule_config)
        elif masking_type == MaskingType.ENCRYPT:
            return self._encrypt(value, rule_config)
        elif masking_type == MaskingType.REPLACE:
            return self._replace(value, rule_config)
        elif masking_type == MaskingType.TRUNCATE:
            return self._truncate(value, rule_config)
        else:
            return value
    
    def _mask(self, value: str, config: Dict) -> str:
        """掩码脱敏"""
        value = str(value)
        start = config.get('start', 0)
        end = config.get('end', len(value))
        mask_char = config.get('mask_char', '*')
        
        if start < 0:
            start = len(value) + start
        if end < 0:
            end = len(value) + end
        
        return value[:start] + mask_char * (end - start) + value[end:]
    
    def _hash(self, value: str, config: Dict) -> str:
        """哈希脱敏"""
        value = str(value)
        algorithm = config.get('algorithm', 'md5')
        salt = config.get('salt', '')
        
        text = value + salt
        
        if algorithm == 'md5':
            return hashlib.md5(text.encode()).hexdigest()
        elif algorithm == 'sha256':
            return hashlib.sha256(text.encode()).hexdigest()
        elif algorithm == 'sha512':
            return hashlib.sha512(text.encode()).hexdigest()
        else:
            return hashlib.md5(text.encode()).hexdigest()
    
    def _encrypt(self, value: str, config: Dict) -> str:
        """加密脱敏（使用AES）"""
        from cryptography.fernet import Fernet
        
        value = str(value)
        key = config.get('key')
        
        if not key:
            raise ValueError("加密密钥不能为空")
        
        f = Fernet(key.encode())
        return f.encrypt(value.encode()).decode()
    
    def _replace(self, value: str, config: Dict) -> str:
        """替换脱敏"""
        value = str(value)
        replace_type = config.get('replace_type', 'random')
        
        if replace_type == 'random':
            # 随机替换为相同长度的字符
            chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
            return ''.join(random.choice(chars) for _ in range(len(value)))
        elif replace_type == 'fixed':
            return config.get('fixed_value', '***')
        elif replace_type == 'pattern':
            pattern = config.get('pattern', r'\d')
            return re.sub(pattern, '*', value)
        else:
            return '***'
    
    def _truncate(self, value: str, config: Dict) -> str:
        """截断脱敏"""
        value = str(value)
        length = config.get('length', 3)
        position = config.get('position', 'start')  # start/end
        
        if position == 'start':
            return value[:length]
        else:
            return value[-length:]
```
