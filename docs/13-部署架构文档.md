## 6. 部署操作手册

### 6.1 首次部署

```bash
#!/bin/bash
# deploy.sh - 首次部署脚本

set -e

NAMESPACE="dmp-prod"
VERSION="${1:-v1.0.0}"

echo "=== 开始部署数据中台系统 ==="
echo "版本: $VERSION"

# 1. 创建Namespace
echo "[1/8] 创建Namespace..."
kubectl apply -f namespace.yaml

# 2. 创建ConfigMap和Secret
echo "[2/8] 创建配置..."
kubectl apply -f configmap.yaml
kubectl apply -f secret.yaml

# 3. 部署存储服务
echo "[3/8] 部署存储服务..."
kubectl apply -f storage/

# 4. 部署数据服务
echo "[4/8] 部署数据服务..."
kubectl apply -f database/

# 5. 等待数据库就绪
echo "[5/8] 等待数据库就绪..."
kubectl wait --for=condition=ready pod -l app=postgres -n $NAMESPACE --timeout=300s

# 6. 执行数据库迁移
echo "[6/8] 执行数据库迁移..."
kubectl run alembic-migration --rm -i --restart=Never \
  --image=registry.example.com/dmp/backend:$VERSION \
  -- alembic upgrade head

# 7. 部署应用服务
echo "[7/8] 部署应用服务..."
kubectl apply -f applications/

# 8. 部署监控
echo "[8/8] 部署监控..."
kubectl apply -f monitoring/

echo "=== 部署完成 ==="
echo "前端访问: https://app.dmp.example.com"
echo "API访问: https://api.dmp.example.com"
echo "监控访问: https://monitoring.dmp.example.com"
```

### 6.2 滚动更新

```bash
#!/bin/bash
# rolling-update.sh - 滚动更新脚本

VERSION="${1:-v1.0.1}"
SERVICE="${2:-dmp-integration-service}"

echo "=== 开始滚动更新 ==="
echo "服务: $SERVICE"
echo "版本: $VERSION"

# 更新镜像
kubectl set image deployment/$SERVICE \
  $SERVICE=registry.example.com/dmp/integration-service:$VERSION \
  -n dmp-prod

# 等待滚动更新完成
kubectl rollout status deployment/$SERVICE -n dmp-prod --timeout=300s

echo "=== 更新完成 ==="
```

### 6.3 回滚操作

```bash
#!/bin/bash
# rollback.sh - 回滚脚本

SERVICE="${1:-dmp-integration-service}"

echo "=== 开始回滚 ==="
echo "服务: $SERVICE"

# 查看历史版本
kubectl rollout history deployment/$SERVICE -n dmp-prod

# 回滚到上一个版本
kubectl rollout undo deployment/$SERVICE -n dmp-prod
```
